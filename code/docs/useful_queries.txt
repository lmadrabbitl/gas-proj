select 
  c.id, p.code as parent, c.code, c.name, c.type, c.description, c.created_at, c.updated_at, c.deactivated_at 
from categories c
left join categories p on c.parent_id = p.id;


select
p.code, c.code as cat, t.date, t.description as desc, t.amount as value, a.code as acc, a2.code as acc2, t.transfer_id as transfer, 
t.is_visible as v, t.created_at
from transactions t
join categories c on c.id = t.category_id
left join categories p on p.id = c.parent_id
join accounts a on a.id = t.account_id
left join accounts a2 on a2.id = t.transfer_account_id
order by date desc, created_at desc
limit 100
;
